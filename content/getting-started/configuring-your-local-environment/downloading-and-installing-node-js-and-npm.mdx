import Parser from 'rss-parser';
import fetch from 'node-fetch';

const parser = new Parser();
const FEEDS = [
  // ใส่ RSS ด้านไอที/แอปเปิล/ไอแพด ที่อนุญาตให้ใช้งาน
  'https://example.com/feed/ipad.xml',
  'https://example.com/feed/tech.xml'
];

const API_BASE = process.env.API_BASE || 'http://localhost:4000/api';
const ADMIN_TOKEN = process.env.ADMIN_TOKEN; // สร้าง token แอดมินแล้วคัดลอกมาใส่ .env

function toSlug(text=''){
  return text.toLowerCase()
    .replace(/\s+/g,'-')
    .replace(/[^\w\-]+/g,'')
    .replace(/\-\-+/g,'-')
    .replace(/^-+|-+$/g,'');
}

async function upsertArticle(item){
  // แปลง item เป็นโครงสร้างบทความ
  const title = item.title?.trim() || 'Untitled';
  const summary = item.contentSnippet?.slice(0, 200) || '';
  const content = item.content || item['content:encoded'] || item.link || '';
  const cover_image = item.enclosure?.url || '';
  const payload = {
    title,
    summary,
    content,
    cover_image,
    category_id: 1,  // สมมติหมวด iPad = 1 (แก้ตามจริง)
    published: 1
  };

  await fetch(`${API_BASE}/articles`, {
    method: 'POST',
    headers: {
      'Content-Type':'application/json',
      'Authorization':`Bearer ${ADMIN_TOKEN}`
    },
    body: JSON.stringify(payload)
  });
}

async function run(){
  for(const url of FEEDS){
    try{
      const feed = await parser.parseURL(url);
      for(const item of feed.items){
        // ใส่เงื่อนไขกันโพสต์ซ้ำจาก link หรือ title (เช่น เก็บแฮชไว้)
        await upsertArticle(item);
      }
    }catch(e){
      console.error('Feed error:', url, e.message);
    }
  }
  console.log('done', new Date().toISOString());
}

run();
